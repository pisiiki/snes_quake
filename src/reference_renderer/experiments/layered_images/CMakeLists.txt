cmake_minimum_required(VERSION 3.25)
project(layered_images LANGUAGES CXX)
# An independent build: the ordinary renderer never sees these sources.
find_package(CLI11 CONFIG REQUIRED)
find_package(SDL3 CONFIG REQUIRED)
find_package(imgui CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
set(REF "${CMAKE_CURRENT_SOURCE_DIR}/../..")
file(GLOB CORE "${REF}/*.cpp")
list(REMOVE_ITEM CORE "${REF}/main.cpp")
add_library(reference_core STATIC ${CORE})
add_executable(layered_images main.cpp capture.cpp reprojection.cpp live_assets.cpp)
foreach(target reference_core layered_images)
  target_include_directories(${target} PRIVATE "${REF}")
  target_compile_features(${target} PRIVATE cxx_std_23)
  target_compile_definitions(${target} PRIVATE SDL_MAIN_HANDLED)
  target_link_libraries(${target} PRIVATE CLI11::CLI11 SDL3::SDL3 imgui::imgui nlohmann_json::nlohmann_json)
  if(MSVC)
    target_compile_options(${target} PRIVATE /Brepro /EHsc /permissive- /utf-8 /W4 /WX)
    target_link_options(${target} PRIVATE /Brepro)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
target_link_libraries(layered_images PRIVATE reference_core)
if(WIN32)
  target_link_libraries(layered_images PRIVATE advapi32)
endif()
